export interface CompositePlayerResult {
  playerKey: string;

  tier: CompositeDataTier;

// Composite Scoring
async calculateCompositeScores(seasonId: string, weights: CompositeWeights): Promise<CompositeScoresResponse> {
  const response = await fetchWithCSRFRetry(
    `${API_BASE_URL}/seasons/${seasonId}/composite-scores/calculate/`,
    { method: 'POST', ...getDefaultOptions(), body: JSON.stringify(transformCompositeWeightsToApi(weights)) }
  );
  const data = await handleResponse(response);
  return {
    players: (data.players || []).map(transformCompositeResultFromApi),
    weights: { coachRating: data.weights.coach_rating, tryoutScore: data.weights.tryout_score, stationWeights: data.weights.station_weights },
    seasonId: data.season_id,
    totalPlayers: data.total_players,
    tierCounts: { full: data.tier_counts.full, coachOnly: data.tier_counts.coach_only, tryoutOnly: data.tier_counts.tryout_only },
  };
}

async listCompositeScores(seasonId: string, weights?: CompositeWeights): Promise<CompositeScoresResponse> {
  let url = `${API_BASE_URL}/seasons/${seasonId}/composite-scores/`;
  if (weights) {
    const params = new URLSearchParams({ coach_rating: String(weights.coachRating), tryout_score: String(weights.tryoutScore) });
    url += `?${params}`;
  }
  const response = await fetch(url, getDefaultOptions());
  const data = await handleResponse(response);
  // Same transform as calculate
}

async getCompositePlayerDetail(seasonId: string, playerKey: string, weights?: CompositeWeights): Promise<CompositePlayerDetail> {
  let url = `${API_BASE_URL}/seasons/${seasonId}/composite-scores/${playerKey}/`;
  if (weights) {
    const params = new URLSearchParams({ coach_rating: String(weights.coachRating), tryout_score: String(weights.tryoutScore) });
    url += `?${params}`;
  }
  const response = await fetch(url, getDefaultOptions());
  const data = await handleResponse(response);
  return transformCompositeDetailFromApi(data);
}

  await fetchWithCSRFRetry(

    { method: 'POST', ...getDefaultOptions(), body: JSON.stringify({ player_key: playerKey, target_tier: targetTier }) }
  );
}

**3. useCompositeScores hook** (`src/hooks/useCompositeScores.ts`)

Follow the useCoachRatings pattern:

```typescript
export function useCompositeScores(seasonId: string | null) {
  const [scores, setScores] = useState<CompositeScoresResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [previousRanks, setPreviousRanks] = useState<Map<string, number>>(new Map());

  const calculate = useCallback(async (weights: CompositeWeights) => {
    if (!seasonId) return;
    setLoading(true);
    setError(null);
    try {
      // Save current ranks before recalculating (for rank delta display)
      if (scores) {
        const rankMap = new Map<string, number>();
        scores.players.forEach((p, i) => rankMap.set(p.playerKey, i));
        setPreviousRanks(rankMap);
      }
      const result = await apiClient.calculateCompositeScores(seasonId, weights);
      setScores(result);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to calculate composite scores');
    } finally {
      setLoading(false);
    }
  }, [seasonId, scores]);

  const getPlayerDetail = useCallback(async (playerKey: string, weights?: CompositeWeights): Promise<CompositePlayerDetail | null> => {
    if (!seasonId) return null;
    try {
      return await apiClient.getCompositePlayerDetail(seasonId, playerKey, weights);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load player detail');
      return null;
    }
  }, [seasonId]);

  const promotePlayer = useCallback(async (playerKey: string, targetTier: CompositeDataTier, currentWeights: CompositeWeights) => {
    if (!seasonId) return;
    try {
      await apiClient.promoteCompositePlayer(seasonId, playerKey, targetTier);
      // Recalculate to refresh rankings
      await calculate(currentWeights);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to promote player');
    }
  }, [seasonId, calculate]);

  return { scores, loading, error, previousRanks, calculate, getPlayerDetail, promotePlayer };
}
```

Key behavior: `previousRanks` stores the rank index of each player BEFORE the last recalculation, enabling the UI to show rank delta arrows.
