```typescript
async createCoachRating(seasonId: string, ratingData: CreateCoachRatingData): Promise<CoachRating> {
  const response = await fetchWithCSRFRetry(
    `${API_BASE_URL}/seasons/${seasonId}/coach-ratings/`,
    {
      ...getDefaultOptions('POST'),
      method: 'POST',
      body: JSON.stringify({
        player_first_name: ratingData.playerFirstName,
        player_last_name: ratingData.playerLastName,
        player_id: ratingData.playerId,
        scores: ratingData.scores,
      }),
    }
  );
  const data = await handleResponse<Record<string, unknown>>(response);
  return transformCoachRatingFromApi(data);
},

async getCoachRating(ratingId: string): Promise<CoachRating> {

    `${API_BASE_URL}/coach-ratings/${ratingId}/`,

  if (updates.playerFirstName !== undefined) body.player_first_name = updates.playerFirstName;

    `${API_BASE_URL}/coach-ratings/${ratingId}/`,
    {
      ...getDefaultOptions('PATCH'),
      method: 'PATCH',
      body: JSON.stringify(body),
    }
  );
  const data = await handleResponse<Record<string, unknown>>(response);
  return transformCoachRatingFromApi(data);
},
```

```typescript
import { useState, useCallback } from 'react';
import type { CoachRating, CoachRatingProgress, CreateCoachRatingData, UpdateCoachRatingData } from '../types';
import { api } from '../api/client';

export function useCoachRatings() {
  const [ratings, setRatings] = useState<CoachRating[]>([]);
  const [progress, setProgress] = useState<CoachRatingProgress[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchRatings = useCallback(async (seasonId: string): Promise<void> => {
    try {
      setIsLoading(true);
      setError(null);
      const data = await api.listCoachRatings(seasonId);
      setRatings(data);
    } catch (err) {
      console.error('Failed to load coach ratings:', err);
      setError('Failed to load coach ratings');
    } finally {
      setIsLoading(false);
    }
  }, []);

  const createRating = useCallback(async (
    seasonId: string,
    data: CreateCoachRatingData
  ): Promise<CoachRating | null> => {
    try {
      setIsLoading(true);
      setError(null);
      const created = await api.createCoachRating(seasonId, data);
      setRatings(prev => [...prev, created]);
      return created;
    } catch (err) {
      console.error('Failed to create coach rating:', err);
      setError('Failed to create coach rating');
      return null;
    } finally {
      setIsLoading(false);
    }
  }, []);

  const updateRating = useCallback(async (
    ratingId: string,
    updates: UpdateCoachRatingData
  ): Promise<CoachRating | null> => {
    try {
      setIsLoading(true);
      setError(null);
      const updated = await api.updateCoachRating(ratingId, updates);
      setRatings(prev => prev.map(r => r.id === updated.id ? updated : r));
      return updated;
    } catch (err) {
      console.error('Failed to update coach rating:', err);
      setError('Failed to update coach rating');
      return null;
    } finally {
      setIsLoading(false);
    }
  }, []);

  const deleteRating = useCallback(async (ratingId: string): Promise<boolean> => {
    try {
      setIsLoading(true);
      setError(null);
      await api.deleteCoachRating(ratingId);
      setRatings(prev => prev.filter(r => r.id !== ratingId));
      return true;
    } catch (err) {
      console.error('Failed to delete coach rating:', err);
      setError('Failed to delete coach rating');
      return false;
    } finally {
      setIsLoading(false);
    }
  }, []);

  const submitRating = useCallback(async (ratingId: string): Promise<CoachRating | null> => {
    try {
      setIsLoading(true);
      setError(null);
      const submitted = await api.submitCoachRating(ratingId);
      setRatings(prev => prev.map(r => r.id === submitted.id ? submitted : r));
      return submitted;
    } catch (err) {
      console.error('Failed to submit coach rating:', err);
      const message = err instanceof Error ? err.message : 'Failed to submit rating';
      setError(message);
      return null;
    } finally {
      setIsLoading(false);
    }
  }, []);

  const fetchProgress = useCallback(async (seasonId: string): Promise<void> => {
    try {
      setIsLoading(true);
      setError(null);
      const data = await api.getCoachRatingsProgress(seasonId);
      setProgress(data);
    } catch (err) {
      console.error('Failed to load rating progress:', err);
      setError('Failed to load rating progress');
    } finally {
      setIsLoading(false);
    }
  }, []);

  return {
    ratings,
    progress,
    isLoading,
    error,
    fetchRatings,
    createRating,
    updateRating,
    deleteRating,
    submitRating,
    fetchProgress,
  };
}
```

```typescript
import { z } from 'zod';
import type { CoachCategoryDefinition } from '../types';

/**
 * Build a Zod schema dynamically from template coach categories.
 * For draft saves, scores are optional (any subset is valid).
 * For submit validation, all categories must have scores.
 */
export function buildRatingSchema(
  categories: CoachCategoryDefinition[],
  requireAll: boolean = false
) {
  const shape: Record<string, z.ZodTypeAny> = {};
  for (const cat of categories) {
    const field = z.number().min(cat.min).max(cat.max);
    shape[cat.id] = requireAll ? field : field.optional();
  }
  return z.object(shape);
}

/**
 * Schema for adding a new player to rate.
 */
export const addPlayerSchema = z.object({
  playerFirstName: z.string().min(1, 'First name is required').max(100),
  playerLastName: z.string().min(1, 'Last name is required').max(100),
});

export type AddPlayerFormData = z.infer<typeof addPlayerSchema>;
```

A reusable 1-5 radio button input for a single coach category. Shows the category name, type badge (physical/soft), and rating buttons with anchor text tooltips.

```typescript
import type { CoachCategoryDefinition } from '../../types';

interface RatingInputProps {
  category: CoachCategoryDefinition;
  value: number | undefined;
  onChange: (value: number) => void;
  disabled?: boolean;
}

/**
 * 1-N rating input with radio buttons and optional anchor labels.
 * Renders a row: category name | [1] [2] [3] [4] [5] buttons
 */
export function RatingInput({ category, value, onChange, disabled }: RatingInputProps) {
  const buttons: number[] = [];
  for (let i = category.min; i <= category.max; i++) {
    buttons.push(i);
  }

  return (
    <div className="flex items-center gap-3 py-2">
      <div className="w-40 flex-shrink-0">
        <span className="text-sm font-medium text-gray-900 dark:text-white">
          {category.name}
        </span>
      </div>
      <div className="flex gap-1.5">
        {buttons.map(num => {
          const isSelected = value === num;
          const anchor = category.anchors?.[String(num)];
          return (
            <button
              key={num}
              type="button"
              disabled={disabled}
              onClick={() => onChange(num)}
              title={anchor || `${num}`}
              className={`w-9 h-9 rounded-lg text-sm font-medium transition-colors ${
                isSelected
                  ? 'bg-blue-600 text-white shadow-sm'
                  : 'bg-gray-100 dark:bg-gray-700 text-gray-700 dark:text-gray-300 hover:bg-gray-200 dark:hover:bg-gray-600'
              } ${disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}`}
            >
              {num}
            </button>
          );
        })}
      </div>
      {/* Show anchor text for selected value */}
      {value !== undefined && category.anchors?.[String(value)] && (
        <span className="text-xs text-gray-500 dark:text-gray-400 italic ml-1">
          {category.anchors[String(value)]}
        </span>
      )}
    </div>
  );
}
```

```typescript
import { useMemo, useCallback } from 'react';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { ArrowLeft, Save, Send, CheckCircle } from 'lucide-react';
import type { CoachRating, CoachCategoryDefinition } from '../../types';
import { buildRatingSchema } from '../../schemas/coachRating';
import { RatingInput } from './RatingInput';

interface PlayerRatingFormProps {
  rating: CoachRating;
  coachCategories: CoachCategoryDefinition[];
  onSaveDraft: (scores: Record<string, number>) => Promise<void>;
  onSubmit: (ratingId: string) => Promise<void>;
  onBack: () => void;
  isSaving?: boolean;
}

export function PlayerRatingForm({
  rating,
  coachCategories,
  onSaveDraft,
  onSubmit,
  onBack,
  isSaving = false,
}: PlayerRatingFormProps) {
  const isSubmitted = rating.status === 'submitted';

  const physicalCategories = useMemo(
    () => coachCategories.filter(c => c.type === 'physical'),
    [coachCategories]
  );
  const softCategories = useMemo(
    () => coachCategories.filter(c => c.type === 'soft'),
    [coachCategories]
  );

  // For draft save: all fields optional. For submit: all required.
  const schema = useMemo(
    () => buildRatingSchema(coachCategories, false),
    [coachCategories]
  );

  const { watch, setValue, handleSubmit, formState: { isDirty } } = useForm({
    resolver: zodResolver(schema),
    defaultValues: rating.scores || {},
  });

  const scores = watch();

  const handleScoreChange = useCallback((categoryId: string, value: number) => {
    setValue(categoryId, value, { shouldDirty: true });
  }, [setValue]);

  const handleDraftSave = useCallback(async () => {
    // Save whatever scores exist (partial is OK for drafts)
    const currentScores: Record<string, number> = {};
    for (const cat of coachCategories) {
      const val = scores[cat.id];
      if (val !== undefined && val !== null) {
        currentScores[cat.id] = val;
      }
    }
    await onSaveDraft(currentScores);
  }, [scores, coachCategories, onSaveDraft]);

  const handleFormSubmit = handleSubmit(async () => {
    // First save current scores, then submit
    const currentScores: Record<string, number> = {};
    for (const cat of coachCategories) {
      const val = scores[cat.id];
      if (val !== undefined && val !== null) {
        currentScores[cat.id] = val;
      }
    }
    await onSaveDraft(currentScores);
    await onSubmit(rating.id);
  });

  // Check if all categories have scores (for enabling submit button)
  const allScored = coachCategories.every(cat => {
    const val = scores[cat.id];
    return val !== undefined && val !== null;
  });

  const renderCategoryGroup = (title: string, categories: CoachCategoryDefinition[]) => {
    if (categories.length === 0) return null;
    return (
      <div className="mb-6">
        <h3 className="text-sm font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider mb-3">
          {title}
        </h3>
        <div className="space-y-1">
          {categories.map(cat => (
            <RatingInput
              key={cat.id}
              category={cat}
              value={scores[cat.id]}
              onChange={(val) => handleScoreChange(cat.id, val)}
              disabled={isSubmitted}
            />
          ))}
        </div>
      </div>
    );
  };

  return (
    <div>
      {/* Header */}
      <div className="flex items-center justify-between mb-6">
        <div className="flex items-center gap-3">
          <button
            onClick={onBack}
            className="p-2 text-gray-500 hover:text-gray-700 dark:text-gray-400 dark:hover:text-gray-200 rounded-lg hover:bg-gray-100 dark:hover:bg-gray-700 transition-colors"
          >
            <ArrowLeft className="w-5 h-5" />
          </button>
          <div>
            <h2 className="text-xl font-semibold text-gray-900 dark:text-white">
              {rating.playerFirstName} {rating.playerLastName}
            </h2>
            {isSubmitted && (
              <span className="inline-flex items-center gap-1 text-sm text-green-600 dark:text-green-400">
                <CheckCircle className="w-4 h-4" />
                Submitted
              </span>
            )}
          </div>
        </div>
      </div>

      {/* Rating Form */}
      <form onSubmit={handleFormSubmit}>
        <div className="bg-white dark:bg-gray-800 rounded-lg border border-gray-200 dark:border-gray-700 p-6">
          {renderCategoryGroup('Physical Skills', physicalCategories)}
          {renderCategoryGroup('Soft Skills', softCategories)}
        </div>

        {/* Actions */}
        {!isSubmitted && (
          <div className="flex items-center justify-end gap-3 mt-4">
            <button
              type="button"
              onClick={handleDraftSave}
              disabled={isSaving || !isDirty}
              className="flex items-center gap-2 px-4 py-2 text-sm font-medium text-gray-700 dark:text-gray-300 bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 rounded-lg hover:bg-gray-50 dark:hover:bg-gray-700 disabled:opacity-50 disabled:cursor-not-allowed transition-colors"
            >
              <Save className="w-4 h-4" />
              Save Draft
            </button>
            <button
              type="submit"
              disabled={isSaving || !allScored}
              className="flex items-center gap-2 px-4 py-2 text-sm font-medium text-white bg-blue-600 rounded-lg hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed transition-colors"
```

```typescript
import { useEffect } from 'react';
import { BarChart3, Users } from 'lucide-react';
import type { CoachRatingProgress } from '../../types';

interface AdminRatingDashboardProps {
  progress: CoachRatingProgress[];
  seasonId: string;
  seasonName: string;
  onFetchProgress: (seasonId: string) => Promise<void>;
  isLoading?: boolean;
}

export function AdminRatingDashboard({
  progress,
  seasonId,
  seasonName,
  onFetchProgress,
  isLoading = false,
}: AdminRatingDashboardProps) {
  useEffect(() => {
    onFetchProgress(seasonId);
  }, [seasonId, onFetchProgress]);

  const totalSubmitted = progress.reduce((sum, p) => sum + p.submittedRatings, 0);
  const totalRatings = progress.reduce((sum, p) => sum + p.totalRatings, 0);

  return (
    <div>
      {/* Header */}

        <BarChart3 className="w-5 h-5 text-gray-500 dark:text-gray-400" />

          </h3>
          <p className="text-sm text-gray-500 dark:text-gray-400">
            {seasonName} — {totalSubmitted}/{totalRatings} ratings submitted
          </p>
        </div>
      </div>

      {isLoading ? (
        <div className="text-center py-8 text-gray-500 dark:text-gray-400">
          Loading progress...
        </div>
      ) : progress.length > 0 ? (
        <div className="bg-white dark:bg-gray-800 rounded-lg border border-gray-200 dark:border-gray-700 overflow-hidden">
          <table className="w-full">
            <thead>
              <tr className="border-b border-gray-200 dark:border-gray-700">

                  Coach
                </th>
                <th className="text-left px-4 py-3 text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                  Email
                </th>
                <th className="text-center px-4 py-3 text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                  Submitted
                </th>
                <th className="text-center px-4 py-3 text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                  Draft
                </th>
                <th className="text-center px-4 py-3 text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                  Total

                <th className="px-4 py-3 text-xs font-semibold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                  Progress

                    <td className="px-4 py-3 text-sm text-center text-yellow-600 dark:text-yellow-400 font-medium">
                      {coach.draftRatings}
                    </td>
                    <td className="px-4 py-3 text-sm text-center text-gray-700 dark:text-gray-300 font-medium">

          <p className="text-gray-500 dark:text-gray-400">
            No ratings have been submitted for this season yet.
          </p>
        </div>
      )}
    </div>
  );
}
```
